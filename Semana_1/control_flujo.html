def clasificar_experimento(accuracy):
    if accuracy >= 0.9:
        return "Excelente"
    elif accuracy >= 0.75:
        return "Aceptable"

def promedio_accuracy(lista_experimentos):
    suma = sum(exp["accuracy"] for exp in lista_experimentos)
    return suma / len(lista_experimentos)
